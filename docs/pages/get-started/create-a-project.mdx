// 📁 app/index.tsx
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import ProfileScreen from '../src/screens/ProfileScreen';
import CheckInScreen from '../src/screens/CheckInScreen';

const Stack = createNativeStackNavigator();

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator>
        <Stack.Screen name="Profile" component={ProfileScreen} />
        <Stack.Screen name="CheckIn" component={CheckInScreen} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}

// 📁 src/screens/ProfileScreen.tsx
import { View, Text, Image, StyleSheet } from 'react-native';
import TrustMeter from '../components/TrustMeter';

const dummyUser = {
  name: 'Jordan Rivers',
  avatar: 'https://via.placeholder.com/100',
  trustScore: 82,
};

export default function ProfileScreen() {
  return (
    <View style={styles.container}>
      <Image source={{ uri: dummyUser.avatar }} style={styles.avatar} />
      <Text style={styles.name}>{dummyUser.name}</Text>
      <TrustMeter score={dummyUser.trustScore} />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { alignItems: 'center', padding: 20 },
  avatar: { width: 100, height: 100, borderRadius: 50 },
  name: { fontSize: 24, marginTop: 10 },
});

// 📁 src/screens/CheckInScreen.tsx
import { View, Text, StyleSheet } from 'react-native';

export default function CheckInScreen() {
  return (
    <View style={styles.container}>
      <Text style={styles.text}>Geo Check-In coming soon...</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  text: { fontSize: 18 },
});

// 📁 src/components/TrustMeter.tsx
import { View, StyleSheet } from 'react-native';

export default function TrustMeter({ score }: { score: number }) {
  const color = score > 80 ? 'green' : score > 50 ? 'orange' : 'red';
  return (
    <View style={styles.wrapper}>
      <View style={[styles.bar, { width: `${score}%`, backgroundColor: color }]} />
    </View>
  );
}

const styles = StyleSheet.create({
  wrapper: { width: '100%', height: 10, backgroundColor: '#ccc', borderRadius: 5 },
  bar: { height: 10, borderRadius: 5 },
});
